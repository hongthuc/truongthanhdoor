    /**
     * Điền nhanh hạng mục vào form báo giá
     */
    function prefillQuote(itemName) {
      const selectEl = document.getElementById('qCategory');
      const noteEl = document.getElementById('qNote');
      if (selectEl) {
        let matched = false;
        for (let i = 0; i < selectEl.options.length; i++) {
          if (itemName.toLowerCase().indexOf(selectEl.options[i].value.toLowerCase()) !== -1) {
            selectEl.selectedIndex = i;
            matched = true;
            break;
          }
        }
        if (!matched && noteEl) {
          noteEl.value = 'Tôi cần nhận báo giá chi tiết cho hạng mục: ' + itemName;
        } else if (noteEl) {
          noteEl.value = 'Quan tâm sản phẩm: ' + itemName;
        }
      }
      showToast('Đã chọn hạng mục: ' + itemName);
    }

    function selectCategoryToast(name) {
      prefillQuote(name);
    }

    let toastTimer = null;
    function showToast(msg) {
      const el = document.getElementById('toastNotice');
      if (!el) return;
      el.textContent = msg;
      el.classList.remove('hidden');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(function () {
        el.classList.add('hidden');
      }, 3000);
    }

    /**
     * Điều khiển 2 Banner tự trượt & hỗ trợ vuốt cảm ứng trên điện thoại
     */
    let currentSlide = 0;
    let slideInterval = null;

    function goToSlide(index) {
      const slides = document.querySelectorAll('.hero-slide-item');
      const dots = document.querySelectorAll('.banner-dot');
      if (!slides.length) return;
      currentSlide = (index + slides.length) % slides.length;

      slides.forEach(function (slide, i) {
        if (i === currentSlide) {
          slide.classList.add('is-active');
        } else {
          slide.classList.remove('is-active');
        }
      });

      dots.forEach(function (dot, i) {
        if (i === currentSlide) {
          dot.classList.add('w-6', 'bg-white');
          dot.classList.remove('w-2.5', 'bg-white/45');
        } else {
          dot.classList.remove('w-6', 'bg-white');
          dot.classList.add('w-2.5', 'bg-white/45');
        }
      });
    }

    function startBannerAutoSlide() {
      clearInterval(slideInterval);
      slideInterval = setInterval(function () {
        goToSlide(currentSlide + 1);
      }, 5000);
    }

      const nextBtn = document.getElementById('bannerNextBtn');
      const sliderEl = document.getElementById('heroSlider');

      if (prevBtn) {
        prevBtn.addEventListener('click', function () {

      if (nextBtn) {
        nextBtn.addEventListener('click', function () {

      // Dừng tự trượt khi rê chuột và hỗ trợ vuốt tay trên mobile
      if (sliderEl) {
        sliderEl.addEventListener('mouseenter', function () { clearInterval(slideInterval); });
        sliderEl.addEventListener('mouseleave', startBannerAutoSlide);

        let touchStartX = 0;
        sliderEl.addEventListener('touchstart', function (e) {
          touchStartX = e.changedTouches[0].screenX;
        }, { passive: true });
        sliderEl.addEventListener('touchend', function (e) {
          const diff = e.changedTouches[0].screenX - touchStartX;
          if (Math.abs(diff) > 40) {
            goToSlide(diff < 0 ? currentSlide + 1 : currentSlide - 1);
            startBannerAutoSlide();
          }
        }, { passive: true });
      }

      // Menu di động
      document.getElementById('mobileMenuOpenBtn').addEventListener('click', openMobileMenu);
      document.getElementById('mobileMenuCloseBtn').addEventListener('click', closeMobileMenu);
      document.getElementById('mobileDrawerBackdrop').addEventListener('click', closeMobileMenu);

      // Đóng modal khi click ra ngoài
      document.getElementById('detailModal').addEventListener('click', function (e) {
        if (e.target === this) closeDetailModal();
      });

      // Xử lý Form yêu cầu báo giá
      const form = document.getElementById('quoteForm');
      const feedback = document.getElementById('quoteFeedback');
      if (form && feedback) {
        form.addEventListener('submit', function (e) {
          e.preventDefault();
          const name = document.getElementById('qName').value.trim();
          const phone = document.getElementById('qPhone').value.trim();
          const category = document.getElementById('qCategory').value;

          if (!name || !phone || phone.replace(/\D/g, '').length < 9) {
            feedback.className = 'mt-4 rounded-lg px-4 py-3 text-xs sm:text-sm font-semibold bg-red-50 text-red-700 border border-red-200';
            feedback.textContent = 'Vui lòng nhập đầy đủ họ tên và số điện thoại hợp lệ để chúng tôi gửi báo giá.';
            feedback.classList.remove('hidden');
            return;
          }

          feedback.className = 'mt-4 rounded-lg px-4 py-3 text-xs sm:text-sm font-semibold bg-emerald-50 text-emerald-800 border border-emerald-200';
          feedback.innerHTML = 'Cảm ơn <strong>' + name + '</strong>! Yêu cầu báo giá <strong>' + category + '</strong> đã được ghi nhận. Kỹ thuật viên Trường Thành Door sẽ gọi lại số <strong>' + phone + '</strong> trong ít phút.';
          feedback.classList.remove('hidden');
          form.reset();
        });
      }

      // Nút cuộn lên đầu trang trên desktop
      const scrollBtn = document.getElementById('scrollTopBtn');
      window.addEventListener('scroll', function () {
        if (!scrollBtn) return;
        if (window.scrollY > 360) {
          scrollBtn.classList.remove('hidden');
          scrollBtn.classList.add('flex');
        } else {
          scrollBtn.classList.add('hidden');
          scrollBtn.classList.remove('flex');
        }
      }, { passive: true });
    });
